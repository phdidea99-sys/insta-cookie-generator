function generateCookies() { const inputData = document.getElementById('inputData').value; if (!inputData.trim()) { alert('❌ প্লিজ কিছু ডাটা লিখুন!'); return; } // বাটন ডিজেবল করুন document.getElementById('submitBtn').disabled = true; document.getElementById('status').innerText = 'প্রসেসিং শুরু হচ্ছে...'; // API কল fetch('/api/submit', { method: 'POST', headers: {'Content-Type': 'application/json'}, body: JSON.stringify({credentials: inputData}) }) .then(response => response.json()) .then(data => { if (data.error) { alert('Error: ' + data.error); return; } // স্ট্যাটাস চেক করা শুরু checkStatus(data.job_id); }) .catch(error => { alert('Error: ' + error); document.getElementById('submitBtn').disabled = false; }); } function checkStatus(jobId) { const interval = setInterval(() => { fetch(`/api/status/${jobId}`) .then(response => response.json()) .then(data => { document.getElementById('progress').style.width = data.progress + '%'; document.getElementById('status').innerText = `প্রসেসিং... ${data.progress}%`; if (data.status === 'completed') { clearInterval(interval); document.getElementById('submitBtn').disabled = false; document.getElementById('status').innerText = 'সম্পন্ন!'; // ডাউনলোড বাটন দেখান document.getElementById('downloadBtn').style.display = 'block'; document.getElementById('downloadBtn').onclick = () => { window.location.href = `/api/download/${jobId}`; }; } }); }, 2000); }
